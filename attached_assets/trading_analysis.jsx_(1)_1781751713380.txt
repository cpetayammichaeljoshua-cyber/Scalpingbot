const MetricCard = ({ label, value, sub, color = "white", small = false }) => (
  <div style={{
    background: CARD, border: `1px solid ${BORDER}`, borderRadius: 12,
    padding: small ? "12px 16px" : "16px 20px", flex: 1, minWidth: small ? 110 : 140
  }}>

    <div style={{ color, fontSize: small ? 18 : 24, fontWeight: 700, lineHeight: 1.1 }}>{value}</div>
    {sub && <div style={{ color: NEUTRAL, fontSize: 11, marginTop: 4 }}>{sub}</div>}
  </div>
);

const SectionTitle = ({ icon, title, subtitle }) => (
  <div style={{ marginBottom: 16 }}>
    <div style={{ display: "flex", alignItems: "center", gap: 8, marginBottom: 4 }}>
      <span style={{ fontSize: 18 }}>{icon}</span>

    </div>
    {subtitle && <p style={{ color: NEUTRAL, fontSize: 12, margin: 0, marginLeft: 26 }}>{subtitle}</p>}
  </div>
);

const Badge = ({ text, color }) => (
  <span style={{
    background: color + "22", border: `1px solid ${color}66`, color,
    fontSize: 10, fontWeight: 700, padding: "2px 8px", borderRadius: 99,
    letterSpacing: "0.05em", textTransform: "uppercase"
  }}>{text}</span>
);

const HourBar = ({ data, field, colorFn, label }) => {
  const max = Math.max(...data.map(d => Math.abs(d[field])));
  return (
    <div style={{ marginBottom: 8 }}>
      <div style={{ color: NEUTRAL, fontSize: 11, marginBottom: 6 }}>{label}</div>
      <div style={{ display: "flex", gap: 1 }}>
        {data.map((d, i) => {
          const val = d[field];
          const h = Math.abs(val) / max * 36;
          const col = colorFn(val, d);
          return (
            <div key={i} style={{ flex: 1, display: "flex", flexDirection: "column", alignItems: "center" }}>
              <div style={{
                width: "100%", height: 36, display: "flex", alignItems: "flex-end", justifyContent: "center"
              }}>
                <div style={{
                  width: "70%", height: h, background: col,
                  borderRadius: "2px 2px 0 0", opacity: 0.85,
                  boxShadow: `0 0 4px ${col}44`
                }} />
              </div>
              <div style={{ color: NEUTRAL, fontSize: 8, marginTop: 2 }}>{d.hour}</div>
            </div>
          );
        })}
      </div>
    </div>
  );
};

const CustomTooltip = ({ active, payload, label, prefix = "", suffix = "" }) => {
  if (!active || !payload?.length) return null;
  return (
    <div style={{ background: "#1e293bee", border: `1px solid ${BORDER}`, borderRadius: 8, padding: "10px 14px", fontSize: 12 }}>
      <div style={{ color: "white", fontWeight: 700, marginBottom: 6 }}>{label}</div>
      {payload.map((p, i) => (
        <div key={i} style={{ color: p.color, marginBottom: 2 }}>
          {p.name}: {prefix}{typeof p.value === "number" ? p.value.toFixed(2) : p.value}{suffix}
        </div>
      ))}
    </div>
  );
};
